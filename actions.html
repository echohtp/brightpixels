Use it in your project Zero new runtime dependencies
import { brightenSurface } from 'brightpixels';
import { bindHold, trackAction } from 'brightpixels/interactions';
import { createParticleEffects } from 'brightpixels/particles';

const light = brightenSurface(card);
const particles = createParticleEffects();
const hold = bindHold(button, {
  surface: light,
  onComplete: () => {
    trackAction(light, save()).then(() => {
      particles.burstFrom(button, { count: 120 });
    }).catch(showError);
  },
});
// On unmount: hold.destroy(); light.destroy(); particles.destroy();
